Just one API call to generate realistic datasets in any format.

import requests

headers = {"X-API-Key": "YOUR_KEY_HERE"}
data = {
    "schema_type": "users",
    "count": 10,
    "locale": "ru_RU"
}

response = requests.post(
    "https://tdd-api-4daf.onrender.com/generate",
    json=data,
    headers=headers
